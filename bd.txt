-- TABLAS DE CATÁLOGOS PRIMERO (Para evitar errores de FK)

CREATE TABLE Roles (
  idRol SERIAL PRIMARY KEY,
  rol VARCHAR(50) NOT NULL
);

CREATE TABLE Tipo_Cita (
  idTipo_Cita SERIAL PRIMARY KEY,
  nombre VARCHAR(50) NOT NULL
);

CREATE TABLE Status_Cita (
  idStatus_Cita SERIAL PRIMARY KEY,
  nombre VARCHAR(50) NOT NULL
);

CREATE TABLE Status_Cotizacion (
  idStatus SERIAL PRIMARY KEY,
  nombre VARCHAR(50) NOT NULL
);

CREATE TABLE Metodos_Pagos (
  idMetodo SERIAL PRIMARY KEY,
  nombre VARCHAR(50) NOT NULL
);

CREATE TABLE Tipo_Pagos (
  idTipo_Pagos SERIAL PRIMARY KEY,
  nombre VARCHAR(50) NOT NULL
);

CREATE TABLE Tipo_Emisor (
  idTipo_Emisor SERIAL PRIMARY KEY,
  nombre VARCHAR(50) NOT NULL
);

-- ESTRUCTURA PRINCIPAL

CREATE TABLE Empresa (
  idEmpresa SERIAL PRIMARY KEY,
  nombre VARCHAR(100),
  telefono VARCHAR(20),
  direccion TEXT,
  ciudad VARCHAR(100),
  estado VARCHAR(100),
  pais VARCHAR(100),
  idPolitica INTEGER
);

CREATE TABLE Cliente (
  idCliente SERIAL PRIMARY KEY,
  nombre VARCHAR(100),
  apellido_paterno VARCHAR(100),
  apellido_materno VARCHAR(100),
  correo VARCHAR(100) UNIQUE,
  contrasenia VARCHAR(255),
  fecha_nacimiento DATE,
  telefono VARCHAR(20)
);

CREATE TABLE Empleado (
  idEmpleado SERIAL PRIMARY KEY,
  idEmpresa INTEGER REFERENCES Empresa(idEmpresa),
  idRol INTEGER REFERENCES Roles(idRol),
  nombre VARCHAR(150),
  correo VARCHAR(100) UNIQUE,
  contrasenia VARCHAR(255),
  fechaNacimiento DATE,
  foto_url TEXT,
  telefono VARCHAR(20)
);

CREATE TABLE Tatuador (
  idTatuador INTEGER PRIMARY KEY,
  porcentajeComision DECIMAL(5,2),
  idEmpleado INTEGER REFERENCES Empleado(idEmpleado)
);

CREATE TABLE Cita (
  idCita SERIAL PRIMARY KEY,
  idTipo_Cita INTEGER REFERENCES Tipo_Cita(idTipo_Cita),
  idStatus_Cita INTEGER REFERENCES Status_Cita(idStatus_Cita),
  idCliente INTEGER REFERENCES Cliente(idCliente),
  idTatuador INTEGER REFERENCES Tatuador(idTatuador),
  fecha DATE,
  hora TIME,
  zonaCuerpo VARCHAR(100),
  tamanio VARCHAR(50), -- Cambiado a VARCHAR para descripciones como "10x10cm"
  idTecnica INTEGER REFERENCES Tecnicas(idTecnica),
  foto_referecncia_url TEXT,
  descripcion TEXT,
  foto_final_url TEXT
);

CREATE TABLE Cotizacion (
  idCotizacion SERIAL PRIMARY KEY,
  idStatus INTEGER REFERENCES Status_Cotizacion(idStatus),
  idCita INTEGER REFERENCES Cita(idCita),
  montoEstimado DECIMAL(10,2),
  precioFinal DECIMAL(10,2),
  justificacion_ajuste TEXT,
  fecha_cotizacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- SEGUIMIENTO Y CHAT

CREATE TABLE Seguimiento (
  idSeguimiento SERIAL PRIMARY KEY,
  idCliente INTEGER REFERENCES Cliente(idCliente),
  idCita INTEGER REFERENCES Cita(idCita),
  idTatuador INTEGER REFERENCES Tatuador(idTatuador),
  estadoCicatrizacion VARCHAR(100),
  estadoSeguimiento VARCHAR(100),
  fechaInicio DATE DEFAULT CURRENT_DATE,
  fechaFin DATE
);

CREATE TABLE Seguimiento_Fotos (
  idSeguimiento_Foto SERIAL PRIMARY KEY,
  idSeguimiento INTEGER REFERENCES Seguimiento(idSeguimiento),
  foto_seguimiento_url TEXT,
  fecha_subida TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE Chat_Seguimiento (
  idMensaje SERIAL PRIMARY KEY,
  idTipo_Emisor INTEGER REFERENCES Tipo_Emisor(idTipo_Emisor),
  idSeguimiento INTEGER REFERENCES Seguimiento(idSeguimiento),
  idEmisor INTEGER, -- ID del Cliente o Empleado según idTipo_Emisor
  mensaje TEXT,
  fecha_envio TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- TABLAS RELACIONALES Y AUXILIARES

CREATE TABLE Cliente_en_Empresa (
  idEmpresa INTEGER REFERENCES Empresa(idEmpresa),
  idCliente INTEGER REFERENCES Cliente(idCliente),
  fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY(idEmpresa, idCliente)
);

CREATE TABLE Horas_Disponibles (
  idHora SERIAL PRIMARY KEY,
  idTatuador INTEGER REFERENCES Tatuador(idTatuador),
  horaInicio TIME,
  horaFin TIME,
  fecha DATE,
  estadodisponibilidad BOOLEAN DEFAULT TRUE,
  motivo TEXT,
  tipo_bloqueo VARCHAR(10), -- 'unico', 'semanal', 'diario'
  dia_semana INTEGER       -- 0=Lunes..6=Domingo (solo para tipo 'semanal')
);

CREATE TABLE Horario_Base (
  id SERIAL PRIMARY KEY,
  idTatuador INTEGER REFERENCES Tatuador(idTatuador),
  dia_semana INTEGER NOT NULL, -- 0=Lunes, 1=Martes, ..., 6=Domingo
  hora_inicio TIME NOT NULL,
  hora_fin TIME NOT NULL,
  activo BOOLEAN DEFAULT TRUE
);

CREATE TABLE Pago (
  idPago SERIAL PRIMARY KEY,
  idMetodo INTEGER REFERENCES Metodos_Pagos(idMetodo),
  idTipo_Pagos INTEGER REFERENCES Tipo_Pagos(idTipo_Pagos),
  idCotizacion INTEGER REFERENCES Cotizacion(idCotizacion),
  idCita INTEGER REFERENCES Cita(idCita),
  monto DECIMAL(10,2),
  fecha DATE DEFAULT CURRENT_DATE
);

CREATE TABLE Precio_Tamano (
  idTamano SERIAL PRIMARY KEY,
  idEmpresa INTEGER REFERENCES Empresa(idEmpresa),
  Tamanio VARCHAR(50),
  altomin DECIMAL(5,2),
  altoMax DECIMAL(5,2),
  anchoMin DECIMAL(5,2),
  anchoMax DECIMAL(5,2),
  precio DECIMAL(10,2)
);

CREATE TABLE Tecnicas (
  idTecnica SERIAL PRIMARY KEY,
  idEmpresa INTEGER REFERENCES Empresa(idEmpresa),
  nombreTecnica VARCHAR(100),
  precio DECIMAL(10,2)
);

CREATE TABLE Tatuador_tiene_Tecnica (
  idTecnica INTEGER REFERENCES Tecnicas(idTecnica),
  idTatuador INTEGER REFERENCES Tatuador(idTatuador),
  PRIMARY KEY(idTecnica, idTatuador)
);

CREATE TABLE observaciones (
  idObserva SERIAL PRIMARY KEY,
  idSeguimiento INTEGER REFERENCES Seguimiento(idSeguimiento),
  idCita INTEGER REFERENCES Cita(idCita),
  observaciones TEXT
);

CREATE TABLE Politica (
    idPolitica SERIAL PRIMARY KEY,
    idEmpresa INTEGER REFERENCES Empresa(idEmpresa),
    titulo VARCHAR(100) NOT NULL,
    contenido TEXT NOT NULL, -- Aquí va todo el texto legal o reglas
    es_activa BOOLEAN DEFAULT TRUE,
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- NOTIFICACIONES PUSH (Web Push API)

-- Tipo_Usuario_Push: 1 = Cliente, 2 = Empleado
CREATE TABLE Push_Subscription (
  idSubscription    SERIAL PRIMARY KEY,
  idTipo_Usuario    INTEGER NOT NULL,           -- 1=Cliente, 2=Empleado (misma semántica que Tipo_Emisor)
  idUsuario         INTEGER NOT NULL,           -- FK lógica a Cliente.idCliente o Empleado.idEmpleado
  idEmpresa         INTEGER REFERENCES Empresa(idEmpresa),
  endpoint          TEXT NOT NULL UNIQUE,       -- URL del push service (FCM, Mozilla, etc.)
  p256dh            TEXT NOT NULL,              -- Clave pública del cliente
  auth              TEXT NOT NULL,              -- Secreto de autenticación
  user_agent        TEXT,                       -- Navegador/dispositivo (opcional, para debug)
  activo            BOOLEAN DEFAULT TRUE,
  fecha_registro    TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_push_sub_usuario ON Push_Subscription(idTipo_Usuario, idUsuario);
CREATE INDEX idx_push_sub_empresa ON Push_Subscription(idEmpresa);

-- Registro de notificaciones enviadas (auditoría + reenvío)
CREATE TABLE Push_Notificacion (
  idNotificacion    SERIAL PRIMARY KEY,
  idSubscription    INTEGER REFERENCES Push_Subscription(idSubscription) ON DELETE CASCADE,
  titulo            VARCHAR(200) NOT NULL,
  cuerpo            TEXT,
  url_destino       TEXT,                       -- Ruta a abrir al hacer clic (ej: /pages/cliente/citas)
  enviada           BOOLEAN DEFAULT FALSE,
  fecha_envio       TIMESTAMP,
  error             TEXT,                       -- Mensaje de error si falló
  fecha_creacion    TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
